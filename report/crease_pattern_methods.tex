\begin{frame}{What this project is}
\begin{columns}[T,onlytextwidth]
\column{0.52\textwidth}
A desktop app that generates, diagnoses, and animates origami crease patterns
on a square sheet — entirely from scratch.
\medskip

\textcolor{cpmid}{\small \texttt{src/cp\_generator/core.py} · \texttt{src/cp\_generator/app.py} · \texttt{src/cp\_generator/fold\_sim.py}}
\medskip

\begin{block}{Five linked algorithms}
\begin{enumerate}\small
  \item Build a planar candidate graph.
  \item Repair parity \& optimize for Kawasaki.
  \item Assign mountains/valleys.
  \item Extract faces \& animate the fold.
  \item Diagnose what is only local, what is globally consistent, and what is still heuristic.
\end{enumerate}
\end{block}

\column{0.44\textwidth}
\textbf{Key objects}
\begin{itemize}\small
  \item Sheet \([0,s]^2\); graph \(G=(V,E)\), positions \(X_v\in\RR^2\).
  \item Partial crease labeling \(\tau:E\to\{U,M,V\}\).
  \item Complete MV assignment \(\sigma:E\to\{-1,+1\}\), with \(+1=M\) and \(-1=V\).
  \item \alert{Sector angle}: angle between consecutive creases at a vertex.
  \item \alert{Half-edge}: oriented edge — determines which face lies to its left.
\end{itemize}
\end{columns}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Mathematical model of the program state}
\begin{columns}[T,onlytextwidth]
\column{0.54\textwidth}
\begin{block}{Current object}
\small
\[
\begin{aligned}
\CP&=(G,X,\tau),\qquad G=(V,E),\\
X&:V\to [0,s]^2,\qquad \tau:E\to\{U,M,V\}.
\end{aligned}
\]
\[
V_{\mathrm{int}}=\{v\in V:X_v\notin \partial[0,s]^2\}.
\]
At each \(v\in V_{\mathrm{int}}\), the incident creases are listed in cyclic
order, with sector angles
\[
\alpha_1(v),\dots,\alpha_{2m(v)}(v),\qquad
\sum_i\alpha_i(v)=2\pi.
\]
\end{block}

\column{0.42\textwidth}
\begin{alertblock}{Two vertexwise quantities}
\small
\[
\kappa(v)=\left|\sum_{i\text{ odd}}\alpha_i(v)-\pi\right|
\]
measures Kawasaki error, while
\[
\mu_\tau(v)=M_\tau(v)-V_\tau(v)
\]
measures Maekawa balance after MV assignment.
\end{alertblock}

\smallskip
\small The pipeline constructs \(G\), improves \(X\), searches for \(\tau\),
then certifies as much global geometry as the code can justify.
\end{columns}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Algorithm I — Planar candidate graph}
\textbf{Goal.} A planar graph dense enough for interesting folds, regular
enough for stable local angle computation.
\medskip

\begin{columns}[T,onlytextwidth]
\column{0.54\textwidth}
\begin{block}{Construction}
\begin{enumerate}\small
  \item Sample interior points; add four corners.
  \item Snap near-boundary points to the boundary.
  \item Compute \(G_0=\operatorname{Del}(\widetilde P)\).
  \item Delete edges lying entirely on one boundary side.
  \item Use the remaining straight-line edges as \(E\).
\end{enumerate}
\end{block}

\column{0.42\textwidth}
\textbf{Mathematical role}\\[3pt]
\small Delaunay does not prove foldability. It supplies a planar candidate
graph with relatively regular local geometry, so the cyclic order and sector
angles used later are numerically stable.
\end{columns}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Algorithm Ib — Generic geometry guard}
\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\begin{block}{Scale-normalized separation}
\small
\[
\delta_{\min}(X)=\min_{u\ne v}\norm{X_u-X_v},
\qquad
\rho(X)=\frac{\delta_{\min}(X)}{s}.
\]
The workflow calls the sample \alert{generic} when
\[
\rho(X)\ge 10^{-3}.
\]
\end{block}

\smallskip
\small The random generator retries until it finds a generic sample, or else
keeps the best available attempt, meaning the one with largest
\(\delta_{\min}(X)\).

\column{0.46\textwidth}
\begin{alertblock}{Why this is separate from Kawasaki}
\small A pattern can satisfy the local angle equations numerically while two
vertices are almost coincident. That makes cyclic-order data unstable and can
poison later exact-face reconstruction.
\end{alertblock}

\smallskip
\small So workflow-level ``local green'' really means
\[
\text{local pass}\ \wedge\ \text{generic geometry}.
\]
\textcolor{cpmid}{\small This is an implemented numerical guard, not a classical origami theorem.}
\end{columns}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Algorithm II — Parity repair \& Kawasaki optimization}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{alertblock}{Necessary condition}
Every interior vertex must have \alert{even degree}:
\(d(v)\equiv 0\pmod 2.\)
\end{alertblock}
\smallskip
\small\textbf{Repair.} Pair odd-degree vertices and delete a shortest path
between each pair. Only the endpoint parities flip.
\textcolor{cpmid}{(Heuristic — not globally optimal.)}

\column{0.48\textwidth}
\begin{block}{Kawasaki's theorem (local) \cite{hull_math_flat}}
Sector angles \(\alpha_1,\dots,\alpha_{2m}\) must satisfy
\[
\alpha_1+\alpha_3+\cdots = \alpha_2+\alpha_4+\cdots = \pi.
\]
\end{block}
\end{columns}

\medskip
\textbf{Geometry repair.} Let \(D(X)=\sum_{v} w_v\,\norm{X_v-X_{0,v}}^2\).
Move vertices as little as possible, subject to the Kawasaki constraints:
\[
\min_{X}\;D(X)+\frac{\alpha}{2}\,D(X)^{2},
\]
with one alternating-angle equality per vertex, since the other follows from
\(\sum_i\alpha_i=2\pi\). Large boundary weights and small interior weights keep
the square nearly fixed. Solved with SLSQP.
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Algorithm III — Local constraints: Maekawa \& Bern--Hayes}
\begin{columns}[T,onlytextwidth]
\column{0.46\textwidth}
\begin{block}{Maekawa's theorem (local) \cite{hull_math_flat}}
At every interior vertex,
\[
M(v)-V(v)=\pm 2.
\]
Used as an \textbf{admissibility filter}: reject any assignment violating
this anywhere.
\end{block}

\column{0.50\textwidth}
\textbf{Bern--Hayes-style reduction \cite{bern_hayes}}\\[4pt]
For each vertex, repeatedly \alert{remove the smallest sector}: its two
bounding creases are forced \emph{equal} or \emph{opposite} in sign.
\smallskip
\begin{enumerate}\small
  \item Sort creases around the vertex by angle.
  \item Find the smallest wedge; record its forced sign relation.
  \item Recurse until the vertex is fully resolved.
\end{enumerate}
\smallskip
\textcolor{cpmid}{\small Boundary vertices get temporary virtual rays to close the cyclic order.}
\end{columns}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Algorithm III — Merging, exact search, \& interlacing}
Each vertex reduction yields sign chains:
\[
\sigma(e_i)=\sigma(e_j)\quad\text{or}\quad\sigma(e_i)=-\sigma(e_j).
\]
\begin{columns}[T,onlytextwidth]
\column{0.52\textwidth}
\textbf{Exact search}
\begin{itemize}\small
  \item Merge chains across shared creases into \(g\) free groups.
  \item Enumerate all \(2^g\) seed assignments; propagate implied signs.
  \item Keep only assignments with \(\abs{M(v)-V(v)}=2\) at every interior vertex.
\end{itemize}
\textcolor{cpmid}{\small Undetermined creases keep label \(-1\); no sign is invented.}

\column{0.44\textwidth}
\begin{block}{Interlacing preference}
\small Among admissible assignments, prefer
\(M,V,M,V,\ldots\) over \(M,M,V,V,\ldots\), weight sharper wedges more,
and subtract a large penalty for the obtuse same--different--same defects on
the next slide.\\[4pt]
\alert{Does not affect correctness.}
\end{block}
\end{columns}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Algorithm IIIb — Repairing obtuse-angle glitches}
\begin{columns}[T,onlytextwidth]
\column{0.54\textwidth}
\begin{block}{Glitch pattern}
\footnotesize For consecutive creases \(e_i,e_{i+1},e_{i+2}\) with consecutive
sector angles \(\beta_i,\beta_{i+1}\),
\[
\beta_i,\beta_{i+1}>\frac{\pi}{2},
\qquad
\sigma(e_i)=\sigma(e_{i+2})\neq \sigma(e_{i+1}),
\]
the code flags a local ``same--different--same'' obstruction across two
consecutive obtuse wedges.
\end{block}

\column{0.42\textwidth}
\begin{alertblock}{Maekawa-preserving fix}
\footnotesize Test exactly two monochromizing moves:
\[
\sigma(e_{i+1})\leftarrow \sigma(e_i)
\quad\text{or}\quad
\sigma(e_i),\sigma(e_{i+2})\leftarrow \sigma(e_{i+1}).
\]
Keep only a move that still satisfies
\[
\abs{M(v)-V(v)}=2
\]
at every interior vertex, then iterate while the total glitch count drops.
\end{alertblock}
\end{columns}

\smallskip
\footnotesize This is a targeted repair for a local/global mismatch: Maekawa
alone may pass even though the resulting flat stack is visually and
combinatorially implausible.
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Algorithm IV — Faces, reflection \& 3D animation}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{Face extraction}\\[4pt]
\small Add the square boundary, order neighbors by polar angle, and trace
half-edges by the left-face rule
\[
(u,v)\mapsto(v,w),
\]
where \(w\) is the predecessor of \(u\) in the cyclic order at \(v\). Each
bounded counterclockwise cycle defines one face.

\medskip
\begin{block}{Reflection rule}
\small If two faces share crease \(\ell\), the child is the parent reflected
in \(\ell\). Propagated from a root face outward; if any cycle is inconsistent,
the method falls back to per-triangle propagation along a spanning tree.
\end{block}

\column{0.48\textwidth}
\textbf{Kinematic animation \cite{abel_rigid,tachi_simulation}}\\[4pt]
Around crease axis \((p,q)\), with sign \(\sigma(e)\in\{\pm 1\}\) and fold
angle \(\theta\):
\[
T_{\text{child}} = R_{(p,q)}\!\bigl(\sigma(e)\,\theta\bigr)\,T_{\text{parent}}.
\]
\small Rigid plates connected by hinge creases: no elasticity, no collisions.
Triangulation is only for rendering, and small normal offsets keep layers
visible near full closure.
\end{columns}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Algorithm IVb — Exact preview motion families}
\begin{columns}[T,onlytextwidth]
\column{0.52\textwidth}
\begin{block}{Legacy layered}
\footnotesize Intuition: keep the original tree pose almost to the end, then
peel layers apart so the final flat stack stays readable.
\[
P_f^{\mathrm{legacy}}(t)
=
\bigl(1-s_{0.72,1}(t)\bigr)P_f^{\mathrm{tree}}(t)+s_{0.72,1}(t)L_f.
\]
\end{block}

\begin{block}{Balanced stack}
\footnotesize Intuition: average the copies of each shared vertex, rigidly
refit whole faces, then blend to a thinner display stack.
\[
\begin{aligned}
Q_f^{\mathrm{bal}}(t)
&=
\mathcal R_{k(t),0.88}(P^{\mathrm{tree}}(t))_f,\\
P_f^{\mathrm{bal}}(t)
&=
\bigl(1-s_{0.76,1}(t)\bigr)Q_f^{\mathrm{bal}}(t)+s_{0.76,1}(t)B_f.
\end{aligned}
\]
\end{block}

\column{0.44\textwidth}
\begin{alertblock}{Seamless rigid}
\footnotesize Intuition: use the exact fold only as a guide, but re-solve a
connected rigid configuration at sample times so seams stay shut and the end
frame is exact.
\[
t_j=\frac{j}{12},
\qquad
j=0,\dots,12.
\]
\footnotesize At each sample:
\[
\min 130\,E_{\mathrm{seam}}
+0.34\,E_{\mathrm{guide}}
+0.75\,E_{\mathrm{hinge}},
\]
\footnotesize then interpolate solved neighbors by SLERP and use nearest-solved
hold instead of the old gappy fallback.
\end{alertblock}
\end{columns}

\smallskip
\footnotesize If exact face reconstruction itself fails, the app still drops to
the guarded triangle-mesh preview.
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Algorithm IVc — What the 3D math actually proves}
\begin{block}{Legacy layered}
\footnotesize For \(t\le0.72\), the renderer is exactly the rigid tree pose.
At \(t=1\), it is exactly the separated stack \(L_f\). The path is continuous,
but the final affine peel-away is generally \alert{not} rigid.
\end{block}

\begin{block}{Balanced stack}
\footnotesize Every relaxation sweep is a rigid fit of the whole face, so the
seam-closing stage preserves each face's internal distances. Already-closed
rigid states are fixed points. The displayed endpoint is the balanced stack
\(B_f\), not the exact folded figure.
\end{block}

\begin{alertblock}{Seamless rigid and mesh fallback}
\footnotesize Every accepted seamless-rigid sample and every SLERP
interpolation between solved neighbors is facewise rigid, with accepted samples
satisfying shared-edge gap \(\le 0.02\). The exact mode starts at the source
shape and ends at the exact folded figure. If exact faces fail entirely, the
mesh fallback still keeps each triangle rigid on its tree pose, but remains
heuristic and warning-level.
\end{alertblock}

\smallskip
\footnotesize Full definitions and proofs are written out in the formal notes.
\end{frame}

\begin{frame}{Algorithm V — Diagnostic status map}
\footnotesize Let \(\CP=(G,X,\tau)\) be the current crease pattern, let \(V_{\mathrm{int}}\subseteq V\) denote the interior vertices, and let
\[
\kappa(v)=\left|\sum_{i\text{ odd}}\alpha_i(v)-\pi\right|
\]
be the Kawasaki residual at \(v\), where \(\alpha_1(v),\dots,\alpha_{2m(v)}(v)\) are the sector angles in cyclic order.

\begin{block}{Reported statuses}
\begin{enumerate}\footnotesize
  \item \textbf{Local status = pass} iff every \(v\in V_{\mathrm{int}}\) has even degree and \(\kappa(v)\le \eps\).
  \item \textbf{Assignment status = pass} iff \(\tau\) is complete and \(|M_\sigma(v)-V_\sigma(v)|=2\) for every \(v\in V_{\mathrm{int}}\).
  \item \textbf{Global status = pass} iff the current embedding has no nonincident crease crossings, every crease borders exactly two traced faces, and reflection propagation is cycle-consistent on the face-adjacency graph.
  \item \textbf{Exact-preview status = pass} iff the current geometry admits exact face reconstruction with no provisional signs and no mesh fallback.
\end{enumerate}
\end{block}

\begin{alertblock}{Non-certifying outcomes}
\footnotesize A warning is reported whenever exact reconstruction needs a reference-geometry reset, provisional completion of \(\tau\), or mesh fallback.
\end{alertblock}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{What each pass actually certifies}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Local and assignment passes}
\footnotesize
\[
\forall v\in V_{\mathrm{int}},\quad
d(v)\in 2\ZZ,\qquad
\kappa(v)\le \eps,
\]
and, after a complete assignment,
\[
\forall v\in V_{\mathrm{int}},\quad
\abs{M_\sigma(v)-V_\sigma(v)}=2.
\]
This is a certificate of \emph{vertexwise admissibility}, not of a global flat
fold.
\end{block}

\column{0.48\textwidth}
\begin{alertblock}{Global and exact-preview passes}
\footnotesize
These certify properties of the \emph{current embedding}: noncrossing creases,
closed traced faces, two-sided face incidence on each fold, and cycle-consistent
reflection propagation, with no provisional MV completion.
\end{alertblock}

\smallskip
\footnotesize The deck therefore separates ``locally admissible'' from
``globally certified current geometry'' as distinct mathematical statements.
\end{columns}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Algorithm VI — Implemented global-consistency test}
\footnotesize The current code does \emph{not} solve arbitrary global flat-foldability. It implements the following certifier on the current embedding \(X\):

\begin{block}{Implemented decision procedure}
\begin{enumerate}\footnotesize
  \item If nonincident fold segments cross, return \(\texttt{fail}\). If there are no folds, return \(\texttt{not\_run}\). If no MV search has been attempted, return \(\texttt{unknown}\).
  \item Augment the fold graph by the square boundary, and order neighbors at each vertex by polar angle.
  \item Trace bounded faces by \((u,v)\mapsto(v,w)\), with \(w\) the predecessor of \(u\) in the cyclic order at \(v\). Reject repetition or nonclosure.
  \item Require every fold edge to border exactly two traced faces. Build the face graph, choose a maximal-area root face, and set \(T_{f_0}=I\).
  \item Across each crease \(\ell\), propagate \(T_g=R_{\ell}T_f\). Hard cycle drift gives \(\texttt{fail}\); soft drift gives \(\texttt{warning}\).
  \item Resolve fold signs from \(\tau\). Missing signs are provisionally filled by face-depth parity, which downgrades to \(\texttt{warning}\).
\end{enumerate}
\end{block}

\begin{alertblock}{Output interpretation}
\footnotesize A final \(\texttt{pass}\) means: no crossings, successful exact
face tracing, two-sided incidence on every fold, cycle-consistent reflections
within tolerance, and no provisional MV completion.
\end{alertblock}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Scope of the local certificates}
\begin{theorem}[Hull's local flat-foldability conditions]
If a single-vertex crease pattern flat-folds, then its degree is even, its sector angles satisfy Kawasaki's alternating-sum identity, and every complete MV assignment satisfies Maekawa's relation \(|M-V|=2\). \textcolor{cpmid}{\cite{hull_math_flat}}
\end{theorem}

\begin{block}{Why this distinction matters}
\footnotesize The implemented \emph{local status} and \emph{assignment status}
test these vertexwise conditions up to numerical tolerance \(\eps\).
Bern--Hayes only derives local sign constraints and shrinks the search over
\(\sigma\); it does not decide global flat-foldability. The deck therefore
keeps ``locally admissible assignment'' and ``globally certified current
geometry'' separate. \textcolor{cpmid}{\cite{bern_hayes}}
\end{block}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Meaning of a global or exact-preview pass}
\begin{theorem}[Implemented exact-current-geometry certificate]
If the algorithm on the previous slide returns \emph{global status = pass} and \emph{exact-preview status = pass}, then the current embedding of \(\CP\) has a noncrossing augmented planar graph, a closed family of bounded traced faces, a two-face incidence relation on every fold edge, and a cycle-consistent family of flat reflections \(T_g=R_{\ell}T_f\), all without provisional completion of the MV data.
\end{theorem}

\begin{block}{Why this theorem matches the code}
\footnotesize
\begin{enumerate}\itemsep2pt
  \item The core checker first rejects nonincident segment crossings.
  \item The fold simulator traces bounded faces by cyclic half-edge traversal and aborts on repetition or nonclosure.
  \item The exact solver requires two incident faces per fold and propagates transforms by explicit crease reflections.
  \item Cycle drift beyond tolerance aborts; provisional sign completion downgrades pass to warning.
\end{enumerate}
\end{block}

\footnotesize This is therefore a theorem about the \emph{current geometry
seen by the code}, not a complete decision procedure for arbitrary multi-vertex
flat-foldability. \textcolor{cpmid}{\cite{hull_zakharevich,eppstein_flat}}
\end{frame}

% ────────────────────────────────────────────────────────────────────────────
\begin{frame}{Algorithm VIb — Bounded exact stack-order checker}
\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\begin{block}{When the checker runs}
\scriptsize Only on exact face reconstructions with
\begin{itemize}\scriptsize
  \item local pass,
  \item complete Maekawa-valid MV assignment,
  \item no crossing folds,
  \item no provisional signs or approximate cycle closure,
  \item and at most \(8\) faces.
\end{itemize}
It samples the near-flat pose at
\[
\theta_{\mathrm{nf}}=\pi-0.08.
\]
\end{block}

\column{0.46\textwidth}
\begin{alertblock}{Order relation on overlaps}
\scriptsize For each overlapping triangle cell \(C\), compute its centroid
\(c_C\), interpolate the two heights \(z_f(c_C)\) and \(z_g(c_C)\), and record
\[
f\succ g\quad\text{or}\quad g\succ f.
\]
If the sign reverses across overlap cells, or the heights become numerically
indistinguishable, the bounded check fails.
\end{alertblock}
\end{columns}

\medskip
\scriptsize The resulting predecessor graph is counted by a bitmask DP over
topological orders, capped at \(10^5\). Pass means at least one bounded stack
order exists; uniqueness means exactly one was found. This is a small-instance
certificate in the overlap-order spirit of \textcolor{cpmid}{\cite{bern_hayes,eppstein_flat}}.
\end{frame}

\begin{frame}{The whole mathematical picture}
\begin{block}{Pipeline as one implication chain}
\scriptsize
\[
\begin{aligned}
\text{sample points}
&\Longrightarrow
\text{planar candidate graph}
\Longrightarrow
\text{generic geometry}
\Longrightarrow
\text{even-degree repair}
\\
&\Longrightarrow
\text{small Kawasaki residual}
\Longrightarrow
\text{locally admissible MV constraints}
\Longrightarrow
\text{exact search over sign groups}
\\
&\Longrightarrow
\text{current assignment }\sigma
\Longrightarrow
\text{face extraction}
\Longrightarrow
\text{reflection compatibility}
\\
&\Longrightarrow
\text{exact or warning-level preview}
\Longrightarrow
\text{bounded stack-order check on small exact cases.}
\end{aligned}
\]
\end{block}

\begin{alertblock}{Rigorous interpretation}
\footnotesize Every arrow above is implemented in the code. The early arrows
are geometric construction and local necessary conditions; the later arrows
certify only the \emph{current geometry}. No claim here is a complete decision
procedure for arbitrary multi-vertex flat-foldability.
\end{alertblock}
\end{frame}

\begin{frame}{Certified today vs.\ unresolved}
\begin{columns}[T,onlytextwidth]
\column{0.46\textwidth}
\begin{block}{Currently certifiable}
\begin{itemize}\small
  \item Vertexwise even-degree and Kawasaki residual checks.
  \item Maekawa verification on a complete assignment \(\sigma\).
  \item Explicit detection of nonincident crease crossings.
  \item Exact face reconstruction on the current geometry.
  \item Bounded stack-order existence for exact instances with at most \(8\) faces.
\end{itemize}
\end{block}

\column{0.50\textwidth}
\begin{block}{Still non-certifying}
\begin{itemize}\small
  \item Parity repair is not globally optimal.
  \item Bern--Hayes reduction is local, not a full global classifier.
  \item Reference-geometry and mesh preview fallbacks remain heuristic.
  \item A true exact global checker is still needed for larger patterns.
\end{itemize}
\end{block}
\end{columns}

\begin{alertblock}{Best theoretical next step}
\footnotesize The most plausible upgrade path is a bounded-instance exact checker based on ply and cell-adjacency complexity, built on the face arrangement already computed by the exact preview solver. \textcolor{cpmid}{\cite{eppstein_flat}}
\end{alertblock}
\end{frame}
